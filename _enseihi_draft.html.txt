    cmp.innerHTML='<div style="grid-column:1/-1;font-size:12px;color:var(--gray);text-align:center;padding:8px;">同一エリアのため比較なし</div>';
  }else{
    ['shinkansen','plane','bus'].forEach(k=>{
      const t=fare[k];
      const d=document.createElement('div');
      d.className='trans-card'+(k===transUsed?' active':'')+(t?'':' na');
      d.innerHTML=`<div class="trans-icon">${TNAME[k].split(' ')[0]}</div><div class="trans-name">${TNAME[k].split(' ')[1]}</div>`+
        (t?`<div class="trans-price" style="font-size:${t.fixed?12:10}px">${t.label}</div><div class="trans-time">${fmtMin(t.min)}</div>`:`<div class="trans-time">—</div>`);
      if(t){d.onclick=()=>{sel.trans=k;document.querySelectorAll('#trans-chips .chip').forEach(c=>c.classList.toggle('active',c.dataset.v===k));calc();};}
      cmp.appendChild(d);
    });
  }

  // Agoda
  document.getElementById('aff-city').textContent=area.name;
  const agodaUrl=`https://www.agoda.com/partners/partnersearch.aspx?cid=${AGODA_CID}&city=${area.agodaCity}`;
  ['aff-link','hero-cta','bd-link'].forEach(id=>document.getElementById(id).href=agodaUrl);
  document.getElementById('hero-cta-city').textContent=area.name;
  document.getElementById('hero-cta').style.display=sel.nights?'block':'none';
  document.getElementById('aff-desc').textContent=sel.nights
    ?`${area.name}のビジネスホテルは約${fmt(area.hotel.biz)}円が目安。試合開催日は球場周辺から埋まるので、早めに実際の価格をチェックしておきましょう。`
    :`日帰りでも試合後に泊まると翌日ゆっくり観光できます。${area.name}の実際のホテル価格をチェックしてみましょう。`;

  document.getElementById('page-input').classList.add('hidden');
  document.getElementById('page-result').classList.remove('hidden');
  window.scrollTo({top:0,behavior:'smooth'});
}

function share(){
  if(!last)return;
  const text=`⚾ 遠征費計算ツール\n${last.from} → ${last.st.name}\n${last.nights?last.nights+'泊':'日帰り'}・${last.people}人で 約${fmt(last.total)}円\n\n自分の遠征も計算してみる👇`;
  if(navigator.share){navigator.share({title:'プロ野球 遠征費計算ツール',text,url:location.href});}
  else{navigator.clipboard.writeText(text+'\n'+location.href).then(()=>alert('コピーしました！'));}
}
